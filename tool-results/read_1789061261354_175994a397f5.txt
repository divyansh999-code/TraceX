   321→   321→  const ready = useRefresh("trends");
   322→   322→  const chartTheme = useChartTheme();

   324→   324→  /* local sheet state — opens on mount when arriving with a topic, then via selectTopic */
   325→   325→  const [sheetOpen, setSheetOpen] = useState(() => selectedTopicId != null);
   326→   326→  const selectTopic = (id: string) => {

   329→   329→  };
   330→   330→  const handleSheetOpenChange = (open: boolean) => {
   331→   331→    setSheetOpen(open);
   332→   332→    if (!open) setSelectedTopicId(null);
   333→   333→  };
   334→   334→
   335→   335→  /* filter-aware topic rows (sorted by 24h change) */
   336→   336→  const topicRows = useMemo(
   337→   337→    () =>
   338→   338→      effectiveTopics(filters)
   339→   339→        .map(({ topic }) => {
   340→   340→          const series = getTopicSeries(topic, filters);
   341→   341→          return { topic, series, volume: series.reduce((a, p) => a + p.total, 0) };
   342→   342→        })
   343→   343→        .sort((a, b) => b.topic.change24h - a.topic.change24h),
   344→   344→    [filters]
   345→   345→  );
   346→   346→
   347→   347→  const activeId = selectedTopicId ?? topicRows[0]?.topic.id ?? null;
   348→   348→  const activeRow = activeId ? (topicRows.find((r) => r.topic.id === activeId) ?? null) : null;
   349→   349→  const activeTopic = activeRow?.topic ?? getTopicById(activeId);

   351→   351→    if (activeRow) return activeRow.series;
   352→   352→    if (activeTopic) return getTopicSeries(activeTopic, filters);
   353→   353→    return [];
   354→   354→  }, [activeRow, activeTopic, filters]);
   355→   355→
   356→   356→  /* drill-down sheet lifecycle */
   357→   357→  const sheetTopic = getTopicById(selectedTopicId);
   358→   358→  const sheetSeries = useMemo(
   359→   359→    () => (sheetTopic ? getTopicSeries(sheetTopic, filters) : []),
   360→   360→    [sheetTopic, filters]
   361→   361→  );
   362→   362→  const sheet24h = useMemo(
   363→   363→    () =>
   364→   364→      sheetTopic
   365→   365→        ? getTopicSeries(sheetTopic, { ...filters, range: "24h" }).reduce((a, p) => a + p.total, 0)
   366→   366→        : 0,
   367→   367→    [sheetTopic, filters]
   368→   368→  );
   369→   369→
   370→   370→  const emerging = topicRows.filter((r) => r.topic.emerging);
   371→   371→  const filteredRows =
   372→   372→    kw.trim() === ""
   373→   373→      ? topicRows
   374→   374→      : topicRows.filter((r) => {
   375→   375→          const hay = `${r.topic.label} ${r.topic.gloss} ${r.topic.category}`.toLowerCase();
   376→   376→          return kw
   377→   377→            .trim()
   378→   378→            .toLowerCase()
   379→   379→            .split(/\s+/)
   380→   380→            .every((tk) => hay.includes(tk.replace(/^[#:]/, "")));
   381→   381→        });

   388→   388→        <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
   389→   389→          <PanelSkeleton className="xl:col-span-5" height="h-64" />
   390→   390→          <PanelSkeleton className="xl:col-span-7" height="h-64" />
   391→   391→        </div>
   392→   392→      </div>
   393→   393→    );
   394→   394→  }
   395→   395→
   396→   396→  const windowLabel = WINDOW_LABEL[filters.range] ?? "window";
   397→   397→
   398→   398→  /* empty filter-bank state */
   399→   399→  if (topicRows.length === 0) {
   400→   400→    return (
   401→   401→      <div className="space-y-4 animate-in fade-in duration-300">
   402→   402→        <ScreenHeader
   403→   403→          kicker="MODULE 02 // TREND EXPLORER"
   404→   404→          title="Trend Explorer"
   405→   405→          description="Narrative velocity against historical baselines, spike detection and emerging-narrative flags across X and Telegram."
   406→   406→          right={<Badge tone="cyan" dot>{filters.platform === "all" ? "X + Telegram" : filters.platform === "x" ? "X only" : "Telegram only"}</Badge>}
   407→   407→        />
   408→   408→        <Panel title="Monitored narratives" icon={SearchX} sub="no results">

   410→   410→            <SearchX className="size-8 text-muted-foreground/60" strokeWidth={1.5} />
   411→   411→            <p className="mt-3 text-sm font-medium text-foreground">
   412→   412→              No narratives match the current filter bank

   414→   414→            <p className="mt-1 text-xs text-muted-foreground max-w-md">
   415→   415→              Adjust the platform, language or search filters in the top bar, or reset the filter
   416→   416→              bank to restore the monitored narrative set.
   417→   417→            </p>
   418→   418→            <Button variant="outline" size="sm" className="mt-4 h-7 text-[11px]" onClick={resetFilters}>

   426→   426→
   427→   427→  const topChips = topicRows.slice(0, 10);
   428→   428→  const moreChips = topicRows.slice(10);
   429→   429→
   430→   430→  /* velocity stats strip */
   431→   431→  const peak = activeSeries.length ? activeSeries.reduce((a, b) => (b.total > a.total ? b : a)) : null;
   432→   432→  const avgTotal = activeSeries.length
   433→   433→    ? activeSeries.reduce((a, p) => a + p.total, 0) / activeSeries.length
   434→   434→    : 0;
   435→   435→  const avgBase = activeSeries.length
   436→   436→    ? activeSeries.reduce((a, p) => a + p.baseline, 0) / activeSeries.length
   437→   437→    : 0;
   438→   438→  const vsBaseline = avgBase > 0 ? (avgTotal / avgBase - 1) * 100 : 0;
   439→   439→  const spikeCount = activeSeries.filter((p) => p.spike).length;
   440→   440→
   441→   441→  return (
   442→   442→    <div className="space-y-4 animate-in fade-in duration-300">
   443→   443→      <ScreenHeader
   444→   444→        kicker="MODULE 02 // TREND EXPLORER"
   445→   445→        title="Trend Explorer"
   446→   446→        description={`Narrative velocity against historical baselines for the ${windowLabel} — spike flags, emerging narratives and per-topic drill-down across X and Telegram.`}
   447→   447→        right={

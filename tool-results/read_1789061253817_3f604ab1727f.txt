   126→
   127→function TopicDrill({
   128→  topic,
   129→  series,
   130→  volume24h,
   131→  onSelectTopic,
   132→}: {
   133→  topic: Topic;
   134→  series: VolumePoint[];
   135→  volume24h: number;
   136→  onSelectTopic: (id: string) => void;
   137→}) {
   138→  const chartTheme = useChartTheme();
   139→  const posts = getSamplePosts(topic.id);
   140→  const related = topic.relatedTopics
   141→    .map((id) => getTopicById(id))
   142→    .filter((t): t is Topic => t !== null);
   143→
   144→  const xVol = series.reduce((a, p) => a + p.x, 0);
   145→  const tgVol = series.reduce((a, p) => a + p.telegram, 0);
   146→  const totalVol = xVol + tgVol;
   147→  const platformData = [
   148→    { name: "X", value: xVol },
   149→    { name: "Telegram", value: tgVol },
   150→  ];
   151→  const velocity = series.slice(-14).map((p) => ({ t: p.t, label: p.label, v: p.total }));
   152→
   153→  const s = topic.sentiment;
   154→  const dom =
   155→    s.positive >= s.neutral && s.positive >= s.negative
   156→      ? "positive"
   157→      : s.negative >= s.neutral
   158→        ? "negative"
   159→        : "neutral";
   160→  const domTone: "green" | "red" | "slate" = dom === "positive" ? "green" : dom === "negative" ? "red" : "slate";
   161→  const langs = topic.languages
   162→    .map((code) => LANGUAGES.find((l) => l.code === code)?.native ?? code)
   163→    .join(" · ");
   164→
   165→  return (
   166→    <>
   167→      <SheetHeader className="border-b border-border pb-3">
   168→        <div className="flex items-center gap-2 flex-wrap pr-8">
   169→          <SheetTitle className="font-display text-lg font-semibold tracking-tight text-foreground">
   170→            {topic.label}
   171→          </SheetTitle>
   172→          <Badge tone="slate">{topic.category}</Badge>
   173→          {topic.emerging && <Badge tone="orange">emerging</Badge>}
   174→        </div>
   175→        <SheetDescription className="text-xs leading-relaxed">{topic.gloss}</SheetDescription>
   176→      </SheetHeader>
   177→
   178→      <div className="p-4 space-y-4">
   179→        {/* telemetry */}
   180→        <div>
   181→          <Taxonomy className="block mb-1">Telemetry</Taxonomy>
   182→          <MetricRow label="Volume · 24h" value={fmtCompact(volume24h)} />
   183→          <MetricRow label="24h change" value={<Delta value={topic.change24h} />} />
   184→          <MetricRow
   185→            label="Dominant sentiment"
   186→            value={
   187→              <Badge tone={domTone} dot>
   188→                {dom}
   189→              </Badge>
   190→            }
   191→          />
   192→          <MetricRow
   193→            label="Risk score"
   194→            value={<span className={riskTextClass(riskTone(topic.risk))}>{fmtScore(topic.risk)}</span>}
   195→          />
   196→          <MetricRow label="Languages" value={langs || "—"} />
   197→        </div>
   198→
   199→        {/* platform split donut + velocity mini chart */}
   200→        <div className="grid grid-cols-2 gap-3">
   201→          <div className="border border-border rounded-md p-3 min-w-0">
   202→            <Taxonomy>Platform split</Taxonomy>
   203→            <div className="relative h-32 mt-1">
   204→              <ResponsiveContainer width="100%" height="100%">
   205→                <PieChart>
   206→                  <Tooltip
   207→                    content={(props) => (
   208→                      <ChartTooltip
   209→                        {...props}
   210→                        format={(e) =>
   211→                          typeof e.value === "number"
   212→                            ? `${((e.value / Math.max(1, totalVol)) * 100).toFixed(0)}%`
   213→                            : String(e.value)
   214→                        }
   215→                      />
   216→                    )}
   217→                  />
   218→                  <Pie data={platformData} dataKey="value" nameKey="name" innerRadius={36} outerRadius={54} strokeWidth={0}>
   219→                    <Cell fill={CHART.orange} />
   220→                    <Cell fill={CHART.green} />
   221→                  </Pie>
   222→                </PieChart>
   223→              </ResponsiveContainer>
   224→              <div className="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
   225→                <span className="font-mono text-sm tnum text-foreground leading-none">
   226→                  {fmtCompact(totalVol)}
   227→                </span>
   228→                <span className="taxonomy text-[9px] text-muted-foreground mt-1">posts</span>
   229→              </div>
   230→            </div>
   231→            <div className="mt-1.5 flex justify-center">
   232→              <Legend
   233→                items={[
   234→                  { label: `X ${Math.round((xVol / Math.max(1, totalVol)) * 100)}%`, color: CHART.orange },
   235→                  { label: `TG ${Math.round((tgVol / Math.max(1, totalVol)) * 100)}%`, color: CHART.green },
   236→                ]}
   237→              />
   238→            </div>
   239→          </div>
   240→
   241→          <div className="border border-border rounded-md p-3 min-w-0">
   242→            <Taxonomy>Velocity</Taxonomy>
   243→            <div className="h-32 -mx-1 mt-1">
   244→              <ResponsiveContainer width="100%" height="100%">
   245→                <AreaChart data={velocity} margin={{ top: 4, right: 4, left: 4, bottom: 0 }}>
   246→                  <defs>
   247→                    <linearGradient id="tvSheetVel" x1="0" y1="0" x2="0" y2="1">
   248→                      <stop offset="0%" stopColor={CHART.orange} stopOpacity={0.45} />
   249→                      <stop offset="100%" stopColor={CHART.orange} stopOpacity={0.1} />
   250→                    </linearGradient>
   251→                  </defs>
   252→                  <XAxis
   253→                    dataKey="t"
   254→                    type="number"
   255→                    domain={["dataMin", "dataMax"]}
   256→                    tickFormatter={(t: number) => {
   257→                      const p = velocity.find((s) => s.t === t);
   258→                      return p?.label ?? "";
   259→                    }}
   260→                    minTickGap={40}
   261→                    tick={{ fontSize: 9, fill: chartTheme.tick, fontFamily: "var(--font-jetbrains), monospace" }}
   262→                    axisLine={{ stroke: chartTheme.grid }}
   263→                    tickLine={false}
   264→                  />
   265→                  <YAxis hide domain={["dataMin", "dataMax"]} />
   266→                  <Tooltip
   267→                    cursor={{ stroke: chartTheme.crosshair, strokeDasharray: "4 4" }}
   268→                    content={(props) => (
   269→                      <ChartTooltip
   270→                        {...props}
   271→                        label={(() => {
   272→                          const p = velocity.find(
   273→                            (s) => s.t === (props as { payload?: { t?: number } }).payload?.t
   274→                          );
   275→                          return p?.label ?? "";
   276→                        })()}
   277→                      />
   278→                    )}
   279→                  />
   280→                  <Area dataKey="v" name="Volume" stroke={CHART.orange} strokeWidth={1.5} fill="url(#tvSheetVel)" />
   281→                </AreaChart>
   282→              </ResponsiveContainer>
   283→            </div>
   284→          </div>
   285→        </div>
   286→
   287→        {/* sample posts */}
   288→        <div>
   289→          <Taxonomy className="block mb-2">Sample posts</Taxonomy>
   290→          <div className="space-y-2">
   291→            {posts.map((p) => (
   292→              <PostCard key={p.id} post={p} />
   293→            ))}
   294→          </div>
   295→        </div>
   296→
   297→        {/* related narratives */}
   298→        <div className="border-t border-border/60 pt-3">
   299→          <Taxonomy className="block mb-2">Related narratives</Taxonomy>
   300→          <div className="flex flex-wrap gap-1.5">
   301→            {related.length > 0 ? (
   302→              related.map((t) => (
   303→                <Chip key={t.id} onClick={() => onSelectTopic(t.id)} title={t.gloss}>
   304→                  {t.label}
   305→                </Chip>
   306→              ))
   307→            ) : (
   308→              <span className="text-[11px] text-muted-foreground">No linked narratives.</span>
   309→            )}
   310→          </div>
   311→        </div>
   312→      </div>
   313→    </>
   314→  );
   315→}

   317→/* ---------------- Main screen ---------------- */
   318→
   319→export function TrendsScreen() {
   320→  const { filters, go, selectedTopicId, setSelectedTopicId, resetFilters } = useApp();
   321→  const ready = useRefresh("trends");
   322→  const chartTheme = useChartTheme();
   323→  const [kw, setKw] = useState("");
   324→  /* local sheet state — opens on mount when arriving with a topic, then via selectTopic */
   325→  const [sheetOpen, setSheetOpen] = useState(() => selectedTopicId != null);
   326→  const selectTopic = (id: string) => {
   327→    setSelectedTopicId(id);
   328→    setSheetOpen(true);
   329→  };
   330→  const handleSheetOpenChange = (open: boolean) => {
   331→    setSheetOpen(open);
   332→    if (!open) setSelectedTopicId(null);
   333→  };
   334→
   335→  /* filter-aware topic rows (sorted by 24h change) */
   336→  const topicRows = useMemo(
   337→    () =>
   338→      effectiveTopics(filters)
   339→        .map(({ topic }) => {
   340→          const series = getTopicSeries(topic, filters);
   341→          return { topic, series, volume: series.reduce((a, p) => a + p.total, 0) };
   342→        })
   343→        .sort((a, b) => b.topic.change24h - a.topic.change24h),
   344→    [filters]
   345→  );
   346→
   347→  const activeId = selectedTopicId ?? topicRows[0]?.topic.id ?? null;
   348→  const activeRow = activeId ? (topicRows.find((r) => r.topic.id === activeId) ?? null) : null;
   349→  const activeTopic = activeRow?.topic ?? getTopicById(activeId);
   350→  const activeSeries = useMemo(() => {
   351→    if (activeRow) return activeRow.series;
   352→    if (activeTopic) return getTopicSeries(activeTopic, filters);
   353→    return [];
   354→  }, [activeRow, activeTopic, filters]);
   355→
   356→  /* drill-down sheet lifecycle */
   357→  const sheetTopic = getTopicById(selectedTopicId);
   358→  const sheetSeries = useMemo(
   359→    () => (sheetTopic ? getTopicSeries(sheetTopic, filters) : []),
   360→    [sheetTopic, filters]
   361→  );
   362→  const sheet24h = useMemo(
   363→    () =>
   364→      sheetTopic
   365→        ? getTopicSeries(sheetTopic, { ...filters, range: "24h" }).reduce((a, p) => a + p.total, 0)
   366→        : 0,
   367→    [sheetTopic, filters]
   368→  );
   369→
   370→  const emerging = topicRows.filter((r) => r.topic.emerging);
   371→  const filteredRows =
   372→    kw.trim() === ""
   373→      ? topicRows
   374→      : topicRows.filter((r) => {
   375→          const hay = `${r.topic.label} ${r.topic.gloss} ${r.topic.category}`.toLowerCase();
   376→          return kw
   377→            .trim()
   378→            .toLowerCase()
   379→            .split(/\s+/)
   380→            .every((tk) => hay.includes(tk.replace(/^[#:]/, "")));
   381→        });
   382→
   383→  if (!ready) {
   384→    return (
   385→      <div className="space-y-4 animate-in fade-in duration-200">
   386→        <PanelSkeleton height="h-14" />
   387→        <PanelSkeleton height="h-80" />
   388→        <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
   389→          <PanelSkeleton className="xl:col-span-5" height="h-64" />
   390→          <PanelSkeleton className="xl:col-span-7" height="h-64" />
   391→        </div>
   392→      </div>
   393→    );
   394→  }
   395→
   396→  const windowLabel = WINDOW_LABEL[filters.range] ?? "window";
   397→
   398→  /* empty filter-bank state */
   399→  if (topicRows.length === 0) {
   400→    return (
   401→      <div className="space-y-4 animate-in fade-in duration-300">
   402→        <ScreenHeader
   403→          kicker="MODULE 02 // TREND EXPLORER"
   404→          title="Trend Explorer"
   405→          description="Narrative velocity against historical baselines, spike detection and emerging-narrative flags across X and Telegram."
   406→          right={<Badge tone="cyan" dot>{filters.platform === "all" ? "X + Telegram" : filters.platform === "x" ? "X only" : "Telegram only"}</Badge>}
   407→        />
   408→        <Panel title="Monitored narratives" icon={SearchX} sub="no results">
   409→          <div className="flex flex-col items-center justify-center py-16 text-center">
   410→            <SearchX className="size-8 text-muted-foreground/60" strokeWidth={1.5} />
   411→            <p className="mt-3 text-sm font-medium text-foreground">
   412→              No narratives match the current filter bank
   413→            </p>
   414→            <p className="mt-1 text-xs text-muted-foreground max-w-md">
   415→              Adjust the platform, language or search filters in the top bar, or reset the filter
   416→              bank to restore the monitored narrative set.
   417→            </p>
   418→            <Button variant="outline" size="sm" className="mt-4 h-7 text-[11px]" onClick={resetFilters}>
   419→              Reset filter bank
   420→            </Button>
   421→          </div>
   422→        </Panel>
   423→      </div>
   424→    );
   425→  }
   426→
   427→  const topChips = topicRows.slice(0, 10);
   428→  const moreChips = topicRows.slice(10);
   429→
   430→  /* velocity stats strip */
   431→  const peak = activeSeries.length ? activeSeries.reduce((a, b) => (b.total > a.total ? b : a)) : null;
   432→  const avgTotal = activeSeries.length
   433→    ? activeSeries.reduce((a, p) => a + p.total, 0) / activeSeries.length
   434→    : 0;
   435→  const avgBase = activeSeries.length
   436→    ? activeSeries.reduce((a, p) => a + p.baseline, 0) / activeSeries.length
   437→    : 0;
   438→  const vsBaseline = avgBase > 0 ? (avgTotal / avgBase - 1) * 100 : 0;
   439→  const spikeCount = activeSeries.filter((p) => p.spike).length;
   440→
   441→  return (
   442→    <div className="space-y-4 animate-in fade-in duration-300">
   443→      <ScreenHeader
   444→        kicker="MODULE 02 // TREND EXPLORER"
   445→        title="Trend Explorer"
   446→        description={`Narrative velocity against historical baselines for the ${windowLabel} — spike flags, emerging narratives and per-topic drill-down across X and Telegram.`}
   447→        right={
   448→          <div className="flex items-center gap-2">
   449→            <Badge tone="cyan" dot>
   450→              {filters.platform === "all" ? "X + Telegram" : filters.platform === "x" ? "X only" : "Telegram only"}
   451→            </Badge>
   452→            <Badge tone="neutral" dot>
   453→              {topicRows.length} narratives
   454→            </Badge>
   455→            <Button variant="outline" size="sm" className="h-7 text-[11px] gap-1.5" onClick={() => go("sentiment")}>
   456→              Sentiment lab <ArrowUpRight className="size-3" />
   457→            </Button>
   458→          </div>
   459→        }
   460→      />
   461→
   462→      {/* topic selector */}
   463→      <Panel
   464→        title="Monitored narratives"
   465→        icon={Hash}
   466→        sub={`${topicRows.length} tracked`}
   467→        bodyClassName="p-3"
   468→        right={<span className="font-mono text-[10px] text-muted-foreground/70 hidden sm:inline">sorted by 24h Δ</span>}
   469→      >
   470→        <div className="flex flex-wrap items-center gap-1.5">
   471→          {topChips.map(({ topic }) => (
   472→            <Chip
   473→              key={topic.id}
   474→              active={topic.id === activeId}
   475→              onClick={() => selectTopic(topic.id)}
   476→              title={topic.gloss}
   477→            >
   478→              {topic.label}
   479→              <span className="ml-1.5 font-mono text-[10px] tnum opacity-70">
   480→                {fmtSigned(topic.change24h)}%
   481→              </span>
   482→            </Chip>
   483→          ))}
   484→          {moreChips.length > 0 && (
   485→            <Select
   486→              value={moreChips.some((r) => r.topic.id === activeId) ? (activeId ?? undefined) : undefined}
   487→              onValueChange={(v) => selectTopic(v)}
   488→            >
   489→              <SelectTrigger
   490→                size="sm"
   491→                className="h-7 rounded-sm text-[11px] gap-1.5 px-2 font-medium text-muted-foreground"
   492→                aria-label="More narratives"
   493→              >
   494→                <SelectValue placeholder={`+${moreChips.length} more`} />
   495→              </SelectTrigger>
   496→              <SelectContent className="rounded-md">
   497→                {moreChips.map(({ topic }) => (
   498→                  <SelectItem key={topic.id} value={topic.id} className="text-xs rounded-sm">
   499→                    {topic.label}
   500→                    <span className="ml-1.5 font-mono text-[10px] tnum text-muted-foreground">
   501→                      {fmtSigned(topic.change24h)}%
   502→                    </span>
   503→                  </SelectItem>
   504→                ))}
   505→              </SelectContent>
   506→            </Select>
   507→          )}
   508→        </div>
   509→      </Panel>
   510→
   511→      {/* main velocity chart */}
   512→      <Panel
   513→        title="Narrative velocity"
   514→        icon={Activity}
   515→        sub={`${activeTopic?.label ?? "—"} · ${windowLabel}`}
   516→        right={
   517→          <Legend
   518→            items={[
   519→              { label: "Volume", color: CHART.orange },
   520→              { label: "Baseline", color: CHART.cyan, dashed: true },
   521→              { label: "Spike flag", color: CHART.amber },
   522→            ]}
   523→          />
   524→        }
   525→      >
   526→        <div className="h-72 md:h-80 -mx-1">
   527→          <ResponsiveContainer width="100%" height="100%">
   528→            <ComposedChart data={activeSeries} margin={{ top: 6, right: 8, left: 0, bottom: 0 }}>
   529→              <defs>
   530→                <linearGradient id="tvVel" x1="0" y1="0" x2="0" y2="1">
   531→                  <stop offset="0%" stopColor={CHART.orange} stopOpacity={0.5} />
   532→                  <stop offset="100%" stopColor={CHART.orange} stopOpacity={0.16} />
   533→                </linearGradient>
   534→              </defs>
   535→              <CartesianGrid stroke={chartTheme.grid} strokeDasharray={GRID.strokeDasharray} vertical={GRID.vertical} />

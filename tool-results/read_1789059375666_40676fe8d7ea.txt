   171→   171→/* ---------------- Network preview (static mini graph) ---------------- */
   172→   172→
   173→   173→function NetworkPreview() {
   174→   174→  const { go } = useApp();
   175→   175→  const { filters } = useApp();
   176→   176→  const { nodes, edges, communities } = useMemo(() => getNetwork(filters), [filters]);
   177→   177→  const influencers = useMemo(() => getInfluencers(filters, 3), [filters]);
   178→   178→
   179→   179→  const W = 360;
   180→   180→  const H = 210;
   181→   181→  const layout = useMemo(() => {
   182→   182→    const top = [...nodes].sort((a, b) => b.pageRank - a.pageRank).slice(0, 16);
   183→   183→    const ids = new Set(top.map((n) => n.id));
   184→   184→    const subEdges = edges.filter((e) => ids.has(e.source) && ids.has(e.target)).slice(0, 26);
   185→   185→    // deterministic radial-ish layout
   186→   186→    const placed = top.map((n, i) => {
   187→   187→      const angle = (i / top.length) * Math.PI * 2 + (n.id.charCodeAt(4) % 10) * 0.06;
   188→   188→      const radius = n.pageRank > 0.06 ? 0.28 : n.pageRank > 0.04 ? 0.55 : 0.82;
   189→   189→      return {
   190→   190→        ...n,
   191→   191→        x: W / 2 + Math.cos(angle) * radius * (W / 2 - 18),
   192→   192→        y: H / 2 + Math.sin(angle) * radius * (H / 2 - 14),
   193→   193→      };
   194→   194→    });
   195→   195→    return { placed, subEdges };
   196→   196→  }, [nodes, edges]);
   197→   197→
   198→   198→  const byId = new Map(layout.placed.map((n) => [n.id, n]));
   199→   199→  const communityColor = (id: string) => communities.find((c) => c.id === id)?.color ?? "#5FA1C4";
   200→   200→
   201→   201→  return (
   202→   202→    <Panel
   203→   203→      title="Influence network"
   204→   204→      sub="preview"
   205→   205→      icon={Share2}
   206→   206→      right={
   207→   207→        <Button
   208→   208→          variant="outline"
   209→   209→          size="sm"
   210→   210→          className="h-6 px-2 text-[10px] gap-1"
   211→   211→          onClick={() => go("network")}
   212→   212→        >
   213→   213→          Open graph <ArrowUpRight className="size-3" />
   214→   214→        </Button>
   215→   215→      }
   216→   216→    >
   217→   217→      <button
   218→   218→        type="button"
   219→   219→        onClick={() => go("network")}
   220→   220→        className="w-full block cursor-pointer group"
   221→   221→        aria-label="Open full network analysis"
   222→   222→      >
   223→   223→        <svg viewBox={`0 0 ${W} ${H}`} className="w-full group-hover:opacity-90 transition-opacity">
   224→   224→          {layout.subEdges.map((e, i) => {
   225→   225→            const a = byId.get(e.source);
   226→   226→            const b = byId.get(e.target);
   227→   227→            if (!a || !b) return null;
   228→   228→            return (
   229→   229→              <line
   230→   230→                key={i}
   231→   231→                x1={a.x}
   232→   232→                y1={a.y}
   233→   233→                x2={b.x}
   234→   234→                y2={b.y}
   235→   235→                stroke="#232838"
   236→   236→                strokeWidth={Math.max(0.6, e.weight / 6)}
   237→   237→              />
   238→   238→            );
   239→   239→          })}
   240→   240→          {layout.placed.map((n) => {
   241→   241→            const r = 3 + Math.sqrt(n.pageRank) * 34;
   242→   242→            const bot = n.botProb > 0.6;
   243→   243→            return (
   244→   244→              <g key={n.id}>
   245→   245→                <circle
   246→   246→                  cx={n.x}
   247→   247→                  cy={n.y}
   248→   248→                  r={r}
   249→   249→                  fill={communityColor(n.communityId)}
   250→   250→                  fillOpacity={0.85}
   251→   251→                  stroke={bot ? CHART.red : "none"}
   252→   252→                  strokeWidth={bot ? 1.4 : 0}
   253→   253→                />
   254→   254→                {n.botProb > 0.6 && (
   255→   255→                  <circle cx={n.x} cy={n.y} r={r + 2.5} fill="none" stroke={CHART.red} strokeOpacity={0.35} />
   256→   256→                )}
   257→   257→              </g>
   258→   258→            );
   259→   259→          })}
   260→   260→        </svg>
   261→   261→      </button>
   262→   262→      <div className="mt-3 space-y-2">
   263→   263→        <Taxonomy>Top influencers</Taxonomy>
   264→   264→        {influencers.map((inf) => (
   265→   265→          <div key={inf.nodeId} className="flex items-center gap-2 text-xs">
   266→   266→            <span className="size-2 rounded-[2px] shrink-0" style={{ background: inf.communityColor }} />
   267→   267→            <span className="font-mono text-[11px] text-foreground truncate">{inf.handle}</span>
   268→   268→            <span className="ml-auto font-mono text-[10px] tnum text-muted-foreground shrink-0">
   269→   269→              PR {inf.pageRank.toFixed(3)}
   270→   270→            </span>
   271→   271→            <span className="font-mono text-[10px] tnum text-muted-foreground shrink-0 w-12 text-right">
   272→   272→              {fmtCompact(inf.reach)}
   273→   273→            </span>
   274→   274→          </div>
   275→   275→        ))}
   276→   276→      </div>
   277→   277→    </Panel>
   278→   278→  );
   279→   279→}

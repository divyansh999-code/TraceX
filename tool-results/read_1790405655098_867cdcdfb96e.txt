    71→/** Strict token-based palette filter (v0.13): every whitespace-separated
    72→ *  query token must occur as a substring of the item value — no more
    73→ *  loose fuzzy matches (searching "tour" used to highlight a narrative
    74→ *  with "transactions…"). Prefix and whole-phrase matches rank higher.
    75→ *
    76→ *  v0.17: values carrying the `nl-intent` marker opt OUT of token scoring.
    77→ *  Natural-language intent items derive their value from the live query
    78→ *  (resolved topics, filter descriptions) — cmdk re-scores mid-keystroke
    79→ *  against the PREVIOUSLY registered value, so a newly-typed token would
    80→ *  score 0, unmount the item before its updated value re-registers, and
    81→ *  dead-lock the group. Opt-out is safe because those groups only mount
    82→ *  once the query itself parses as the intent — they can never leak into
    83→ *  unrelated searches. */
    84→const tokenFilter: (value: string, search: string) => number = (value, search) => {
    85→  const v = value.toLowerCase();
    86→  if (v.startsWith("nl-intent")) return 1;
    87→  const q = search.toLowerCase().trim();
    88→  if (!q) return 1;
    89→  const tokens = q.split(/\s+/).filter(Boolean);
    90→  if (!tokens.every((t) => v.includes(t))) return 0;
    91→  let score = 1;
    92→  if (v.includes(q)) score += 50; // whole phrase
    93→  if (v.startsWith(tokens[0])) score += 100; // leading token
    94→  return score;
    95→};

    97→/* ---- natural-language intents (v0.16 → v0.17) ----

   105→const FILTER_RE = /^\s*(?:filter|set|use)\s+(.+?)\s*$/i;
   106→const WATCH_RE = /^\s*(?:watch|star|follow)\s+(.+?)\s*$/i;

   138→/** Parse a filter intent into concrete filter-bank writes. Composable:
   139→   "filter x 30d", "filter telegram", "filter hindi last 14 days". */
   140→interface FilterIntent {
   141→  platform?: Platform;
   142→  range?: RangeKey;
   143→  customDays?: number;
   144→  language?: string;
   145→  describe: string[];
   146→}
   147→const parseFilterIntent = (q: string): FilterIntent | null => {
   148→  const s = q.trim().toLowerCase();
   149→  if (!s) return null;
   150→  const intent: FilterIntent = { describe: [] };
   151→
   152→  /* platform tokens */
   153→  if (/\b(?:x|twitter|tweets?)\b/.test(s)) {
   154→    intent.platform = "x";
   155→    intent.describe.push("X only");
   156→  } else if (/\b(?:telegram|tg)\b/.test(s)) {
   157→    intent.platform = "telegram";
   158→    intent.describe.push("Telegram only");
   159→  } else if (/\b(?:all|both|everything)\b/.test(s)) {
   160→    intent.platform = "all";
   161→    intent.describe.push("all platforms");
   162→  }
   163→
   164→  /* range tokens */
   165→  const lastDays = s.match(/last\s+(\d+)\s*days?/);
   166→  if (/\b(?:24h|today|day)\b/.test(s)) {
   167→    intent.range = "24h";
   168→    intent.describe.push("24h window");
   169→  } else if (/\b(?:7d|week)\b/.test(s)) {
   170→    intent.range = "7d";
   171→    intent.describe.push("7d window");
   172→  } else if (/\b(?:30d|month)\b/.test(s)) {
   173→    intent.range = "30d";
   174→    intent.describe.push("30d window");
   175→  } else if (lastDays) {
   176→    const d = Math.min(90, Math.max(3, parseInt(lastDays[1], 10)));
   177→    intent.range = "custom";
   178→    intent.customDays = d;
   179→    intent.describe.push(`${d}d window`);
   180→  }
   181→
   182→  /* language tokens — first named language wins */
   183→  const LANG_ALIASES: Record<string, string> = {
   184→    hindi: "hi",
   185→    हिन्दी: "hi",
   186→    english: "en",
   187→    hinglish: "hinglish",
   188→    tamil: "ta",
   189→    தமிழ்: "ta",
   190→    telugu: "te",
   191→    తెలుగు: "te",
   192→    bengali: "bn",
   193→    bangla: "bn",
   194→    বাংলা: "bn",
   195→    marathi: "mr",
   196→    मराठी: "mr",
   197→  };
   198→  for (const [alias, code] of Object.entries(LANG_ALIASES)) {
   199→    if (new RegExp(`\\b${alias}\\b`).test(s)) {
   200→      intent.language = code;
   201→      intent.describe.push(`${LANGUAGES.find((l) => l.code === code)?.label ?? alias} filter`);
   202→      break;
   203→    }
   204→  }
   205→
   206→  return intent.describe.length ? intent : null;
   207→};

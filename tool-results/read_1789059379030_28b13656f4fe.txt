   171→/* ---------------- Network preview (static mini graph) ---------------- */
   172→
   173→function NetworkPreview() {
   174→  const { go } = useApp();
   175→  const { filters } = useApp();
   176→  const { nodes, edges, communities } = useMemo(() => getNetwork(filters), [filters]);
   177→  const influencers = useMemo(() => getInfluencers(filters, 3), [filters]);
   178→
   179→  const W = 360;
   180→  const H = 210;
   181→  const layout = useMemo(() => {
   182→    const top = [...nodes].sort((a, b) => b.pageRank - a.pageRank).slice(0, 16);
   183→    const ids = new Set(top.map((n) => n.id));
   184→    const subEdges = edges.filter((e) => ids.has(e.source) && ids.has(e.target)).slice(0, 26);
   185→    // deterministic radial-ish layout
   186→    const placed = top.map((n, i) => {
   187→      const angle = (i / top.length) * Math.PI * 2 + (n.id.charCodeAt(4) % 10) * 0.06;
   188→      const radius = n.pageRank > 0.06 ? 0.28 : n.pageRank > 0.04 ? 0.55 : 0.82;
   189→      return {
   190→        ...n,
   191→        x: W / 2 + Math.cos(angle) * radius * (W / 2 - 18),
   192→        y: H / 2 + Math.sin(angle) * radius * (H / 2 - 14),
   193→      };
   194→    });
   195→    return { placed, subEdges };
   196→  }, [nodes, edges]);
   197→
   198→  const byId = new Map(layout.placed.map((n) => [n.id, n]));
   199→  const communityColor = (id: string) => communities.find((c) => c.id === id)?.color ?? "#5FA1C4";
   200→
   201→  return (
   202→    <Panel
   203→      title="Influence network"
   204→      sub="preview"
   205→      icon={Share2}
   206→      right={
   207→        <Button
   208→          variant="outline"
   209→          size="sm"
   210→          className="h-6 px-2 text-[10px] gap-1"
   211→          onClick={() => go("network")}
   212→        >
   213→          Open graph <ArrowUpRight className="size-3" />
   214→        </Button>
   215→      }
   216→    >
   217→      <button
   218→        type="button"
   219→        onClick={() => go("network")}
   220→        className="w-full block cursor-pointer group"
   221→        aria-label="Open full network analysis"
   222→      >
   223→        <svg viewBox={`0 0 ${W} ${H}`} className="w-full group-hover:opacity-90 transition-opacity">
   224→          {layout.subEdges.map((e, i) => {
   225→            const a = byId.get(e.source);
   226→            const b = byId.get(e.target);
   227→            if (!a || !b) return null;
   228→            return (
   229→              <line
   230→                key={i}
   231→                x1={a.x}
   232→                y1={a.y}
   233→                x2={b.x}
   234→                y2={b.y}
   235→                stroke="#232838"
   236→                strokeWidth={Math.max(0.6, e.weight / 6)}
   237→              />
   238→            );
   239→          })}
   240→          {layout.placed.map((n) => {
   241→            const r = 3 + Math.sqrt(n.pageRank) * 34;
   242→            const bot = n.botProb > 0.6;
   243→            return (
   244→              <g key={n.id}>
   245→                <circle
   246→                  cx={n.x}
   247→                  cy={n.y}
   248→                  r={r}
   249→                  fill={communityColor(n.communityId)}
   250→                  fillOpacity={0.85}
   251→                  stroke={bot ? CHART.red : "none"}
   252→                  strokeWidth={bot ? 1.4 : 0}
   253→                />
   254→                {n.botProb > 0.6 && (
   255→                  <circle cx={n.x} cy={n.y} r={r + 2.5} fill="none" stroke={CHART.red} strokeOpacity={0.35} />
   256→                )}
   257→              </g>
   258→            );
   259→          })}
   260→        </svg>
   261→      </button>
   262→      <div className="mt-3 space-y-2">
   263→        <Taxonomy>Top influencers</Taxonomy>
   264→        {influencers.map((inf) => (
   265→          <div key={inf.nodeId} className="flex items-center gap-2 text-xs">
   266→            <span className="size-2 rounded-[2px] shrink-0" style={{ background: inf.communityColor }} />
   267→            <span className="font-mono text-[11px] text-foreground truncate">{inf.handle}</span>
   268→            <span className="ml-auto font-mono text-[10px] tnum text-muted-foreground shrink-0">
   269→              PR {inf.pageRank.toFixed(3)}
   270→            </span>
   271→            <span className="font-mono text-[10px] tnum text-muted-foreground shrink-0 w-12 text-right">
   272→              {fmtCompact(inf.reach)}
   273→            </span>
   274→          </div>
   275→        ))}
   276→      </div>
   277→    </Panel>
   278→  );
   279→}

   547→            </ResponsiveContainer>
   548→          </div>
   549→          <div className="mt-3 grid grid-cols-3 gap-3 border-t border-border/60 pt-3">
   550→            <div>
   551→              <Taxonomy>Peak hour</Taxonomy>
   552→              <div className="font-mono text-sm tnum text-foreground mt-0.5">
   553→                {(() => {
   554→                  const peak = series.reduce((a, b) => (b.total > a.total ? b : a), series[0]);
   555→                  return peak ? `${peak.label} · ${fmtCompact(peak.total)}` : "—";
   556→                })()}
   557→              </div>
   558→            </div>
   559→            <div>
   560→              <Taxonomy>Negative share</Taxonomy>
   561→              <div className="font-mono text-sm tnum text-signal-red mt-0.5">
   562→                {(() => {
   563→                  const tot = series.reduce((a, p) => a + p.total, 0) || 1;
   564→                  const neg = series.reduce((a, p) => a + p.negative, 0);
   565→                  return `${((neg / tot) * 100).toFixed(1)}%`;
   566→                })()}
   567→              </div>
   568→            </div>
   569→            <div>
   570→              <Taxonomy>Spikes flagged</Taxonomy>
   571→              <div className="font-mono text-sm tnum text-signal-amber mt-0.5">
   572→                {series.filter((p) => p.spike).length} vs baseline
   573→              </div>
   574→            </div>
   575→          </div>
   576→        </Panel>
   577→
   578→        <div className="xl:col-span-4 min-w-0">
   579→          <LiveAlertFeed />
   580→        </div>
   581→      </div>
   582→
   583→      {/* Trending narratives + network preview */}
   584→      <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
   585→        <Panel
   586→          className="xl:col-span-7"
   587→          title="Top trending narratives"
   588→          icon={Flame}
   589→          sub={`24h Δ · ${windowLabel}`}
   590→          bodyClassName="p-0"
   591→          right={
   592→            <Button variant="outline" size="sm" className="h-6 px-2 text-[10px] gap-1" onClick={() => go("trends")}>
   593→              Trend explorer <ArrowUpRight className="size-3" />
   594→            </Button>
   595→          }
   596→        >
   597→          <div className="overflow-x-auto">
   598→            <table className="w-full text-left">
   599→              <thead>
   600→                <tr className="border-b border-border">
   601→                  {["Narrative", "Volume", "24h Δ", "Sentiment", "Risk", "Trend"].map((h) => (
   602→                    <th key={h} className="taxonomy text-muted-foreground/70 font-semibold px-4 py-2 whitespace-nowrap">
   603→                      {h}
   604→                    </th>
   605→                  ))}
   606→                </tr>
   607→              </thead>
   608→              <tbody>
   609→                {topics.map(({ topic, spark }) => {

   611→                    topic.sentiment.positive > topic.sentiment.negative

   614→                        : "neutral"
   615→                      : topic.sentiment.negative > topic.sentiment.neutral
   616→                        ? "negative"
   617→                        : "neutral";
   618→                  return (
   619→                    <tr
   620→                      key={topic.id}
   621→                      onClick={() => go("trends", { topicId: topic.id })}
   622→                      className="border-b border-border/50 last:border-0 hover:bg-accent cursor-pointer transition-colors"
   623→                    >
   624→                      <td className="px-4 py-2.5">
   625→                        <div className="flex items-center gap-2 min-w-0">
   626→                          {topic.velocity === "surging" && <Flame className="size-3 text-signal-orange shrink-0" />}
   627→                          <span className="text-xs font-medium text-foreground truncate max-w-40">{topic.label}</span>
   628→                          {topic.emerging && <Badge tone="orange">emerging</Badge>}
   629→                        </div>
   630→                        <div className="text-[10px] text-muted-foreground truncate max-w-44 mt-0.5">{topic.gloss}</div>
   631→                      </td>
   632→                      <td className="px-4 py-2.5 font-mono text-xs tnum text-foreground whitespace-nowrap">
   633→                        {fmtCompact(topic.baseVolume * 1.75)}
   634→                      </td>
   635→                      <td className="px-4 py-2.5 whitespace-nowrap">
   636→                        <Delta value={topic.change24h} />
   637→                      </td>
   638→                      <td className="px-4 py-2.5">
   639→                        <Badge tone={dom === "positive" ? "green" : dom === "negative" ? "red" : "slate"}>
   640→                          {dom}
   641→                        </Badge>
   642→                      </td>
   643→                      <td className="px-4 py-2.5 w-28">
   644→                        <ScoreBar value={topic.risk} showValue />
   645→                      </td>
   646→                      <td className="px-4 py-2.5">
   647→                        <Sparkline data={spark} color={CHART.orange} width={72} height={22} />
   648→                      </td>
   649→                    </tr>
   650→                  );
   651→                })}
   652→              </tbody>
   653→            </table>
   654→          </div>
   655→        </Panel>
   656→
   657→        <div className="xl:col-span-5 min-w-0">
   658→          <NetworkPreview />

   660→      </div>

   662→      {/* Module pipeline */}

   664→        <ModuleStrip />

   667→      <div className="flex items-center gap-2 text-[10px] font-mono text-muted-foreground/60 px-1">

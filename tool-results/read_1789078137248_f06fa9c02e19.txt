   129→/* ---------------- Network preview (static mini graph) ---------------- */
   130→
   131→function NetworkPreview() {
   132→  const { go } = useApp();
   133→  const { filters } = useApp();
   134→  const { nodes, edges, communities } = useMemo(() => getNetwork(filters), [filters]);
   135→  const influencers = useMemo(() => getInfluencers(filters, 3), [filters]);
   136→
   137→  const W = 360;
   138→  const H = 210;
   139→  const layout = useMemo(() => {
   140→    const top = [...nodes].sort((a, b) => b.pageRank - a.pageRank).slice(0, 16);
   141→    const ids = new Set(top.map((n) => n.id));
   142→    const subEdges = edges.filter((e) => ids.has(e.source) && ids.has(e.target)).slice(0, 26);
   143→    // deterministic radial-ish layout
   144→    const placed = top.map((n, i) => {
   145→      const angle = (i / top.length) * Math.PI * 2 + (n.id.charCodeAt(4) % 10) * 0.06;
   146→      const radius = n.pageRank > 0.06 ? 0.28 : n.pageRank > 0.04 ? 0.55 : 0.82;
   147→      return {
   148→        ...n,
   149→        x: W / 2 + Math.cos(angle) * radius * (W / 2 - 18),
   150→        y: H / 2 + Math.sin(angle) * radius * (H / 2 - 14),
   151→      };
   152→    });
   153→    return { placed, subEdges };
   154→  }, [nodes, edges]);
   155→
   156→  const byId = new Map(layout.placed.map((n) => [n.id, n]));
   157→  const communityColor = (id: string) => communities.find((c) => c.id === id)?.color ?? "#5FA1C4";
   158→
   159→  return (
   160→    <Panel
   161→      title="Influence network"
   162→      sub="preview"
   163→      icon={Share2}
   164→      right={
   165→        <Button
   166→          variant="outline"
   167→          size="sm"
   168→          className="h-6 px-2 text-[10px] gap-1"
   169→          onClick={() => go("network")}
   170→        >
   171→          Open graph <ArrowUpRight className="size-3" />
   172→        </Button>
   173→      }
   174→    >
   175→      <button
   176→        type="button"
   177→        onClick={() => go("network")}
   178→        className="w-full block cursor-pointer group"
   179→        aria-label="Open full network analysis"
   180→      >
   181→        <svg viewBox={`0 0 ${W} ${H}`} className="w-full group-hover:opacity-90 transition-opacity">
   182→          {layout.subEdges.map((e, i) => {
   183→            const a = byId.get(e.source);
   184→            const b = byId.get(e.target);
   185→            if (!a || !b) return null;
   186→            return (
   187→              <line
   188→                key={i}
   189→                x1={a.x}
   190→                y1={a.y}
   191→                x2={b.x}
   192→                y2={b.y}
   193→                stroke="#232838"
   194→                strokeWidth={Math.max(0.6, e.weight / 6)}
   195→              />
   196→            );
   197→          })}
   198→          {layout.placed.map((n) => {
   199→            const r = 3 + Math.sqrt(n.pageRank) * 34;
   200→            const bot = n.botProb > 0.6;
   201→            return (
   202→              <g key={n.id}>
   203→                <circle
   204→                  cx={n.x}
   205→                  cy={n.y}
   206→                  r={r}
   207→                  fill={communityColor(n.communityId)}
   208→                  fillOpacity={0.85}
   209→                  stroke={bot ? CHART.red : "none"}
   210→                  strokeWidth={bot ? 1.4 : 0}
   211→                />
   212→                {n.botProb > 0.6 && (
   213→                  <circle cx={n.x} cy={n.y} r={r + 2.5} fill="none" stroke={CHART.red} strokeOpacity={0.35} />
   214→                )}
   215→              </g>
   216→            );
   217→          })}
   218→        </svg>
   219→      </button>
   220→      <div className="mt-3 space-y-2">
   221→        <Taxonomy>Top influencers</Taxonomy>
   222→        {influencers.map((inf) => (
   223→          <div key={inf.nodeId} className="flex items-center gap-2 text-xs">
   224→            <span className="size-2 rounded-[2px] shrink-0" style={{ background: inf.communityColor }} />
   225→            <span className="font-mono text-[11px] text-foreground truncate">{inf.handle}</span>
   226→            <span className="ml-auto font-mono text-[10px] tnum text-muted-foreground shrink-0">
   227→              PR {inf.pageRank.toFixed(3)}
   228→            </span>
   229→            <span className="font-mono text-[10px] tnum text-muted-foreground shrink-0 w-12 text-right">
   230→              {fmtCompact(inf.reach)}
   231→            </span>
   232→          </div>
   233→        ))}
   234→      </div>
   235→    </Panel>
   236→  );
   237→}

   523→                  .filter((p) => p.spike)
   524→                  .map((p) => (
   525→                    <ReferenceDot
   526→                      key={p.t}
   527→                      x={p.t}
   528→                      y={p.total}
   529→                      r={4}
   530→                      fill={CHART.amber}
   531→                      stroke="#0A0E14"
   532→                      strokeWidth={1}
   533→                    />
   534→                  ))}
   535→              </AreaChart>
   536→            </ResponsiveContainer>
   537→          </div>
   538→          <div className="mt-3 grid grid-cols-3 gap-3 border-t border-border/60 pt-3">
   539→            <div>
   540→              <Taxonomy>Peak hour</Taxonomy>
   541→              <div className="font-mono text-sm tnum text-foreground mt-0.5">
   542→                {(() => {
   543→                  const peak = series.reduce((a, b) => (b.total > a.total ? b : a), series[0]);
   544→                  return peak ? `${peak.label} · ${fmtCompact(peak.total)}` : "—";
   545→                })()}
   546→              </div>
   547→            </div>
   548→            <div>
   549→              <Taxonomy>Negative share</Taxonomy>
   550→              <div className="font-mono text-sm tnum text-signal-red mt-0.5">
   551→                {(() => {
   552→                  const tot = series.reduce((a, p) => a + p.total, 0) || 1;
   553→                  const neg = series.reduce((a, p) => a + p.negative, 0);
   554→                  return `${((neg / tot) * 100).toFixed(1)}%`;
   555→                })()}
   556→              </div>
   557→            </div>
   558→            <div>
   559→              <Taxonomy>Spikes flagged</Taxonomy>
   560→              <div
   561→                className={cn(
   562→                  "font-mono text-sm tnum mt-0.5",
   563→                  series.filter((p) => p.spike).length > 0 ? "text-signal-amber" : "text-muted-foreground"
   564→                )}
   565→              >
   566→                {series.filter((p) => p.spike).length > 0
   567→                  ? `${series.filter((p) => p.spike).length} vs baseline`
   568→                  : "none · nominal"}
   569→              </div>
   570→            </div>
   571→          </div>
   572→        </Panel>
   573→
   574→        <div className="xl:col-span-4 min-w-0">
   575→          <LiveAlertFeed />
   576→        </div>
   577→      </div>
   578→
   579→      {/* Trending narratives + network preview */}
   580→      <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
   581→        <Panel
   582→          className="xl:col-span-7"
   583→          title="Top trending narratives"
   584→          icon={Flame}
   585→          sub={`24h Δ · ${windowLabel}`}
   586→          bodyClassName="p-0"
   587→          right={
   588→            <Button variant="outline" size="sm" className="h-6 px-2 text-[10px] gap-1" onClick={() => go("trends")}>
   589→              Trend explorer <ArrowUpRight className="size-3" />
   590→            </Button>
   591→          }
   592→        >
   593→          <div className="overflow-x-auto">
   594→            <table className="w-full text-left">
   595→              <thead>
   596→                <tr className="border-b border-border">
   597→                  {["Narrative", "Volume", "24h Δ", "Sentiment", "Risk", "Trend"].map((h) => (
   598→                    <th key={h} className="taxonomy text-muted-foreground/70 font-semibold px-4 py-2 whitespace-nowrap">
   599→                      {h}
   600→                    </th>
   601→                  ))}
   602→                </tr>
   603→              </thead>
   604→              <tbody>
   605→                {topics.map(({ topic, spark }) => {
   606→                  const dom =
   607→                    topic.sentiment.positive > topic.sentiment.negative
   608→                      ? topic.sentiment.positive > topic.sentiment.neutral
   609→                        ? "positive"
   610→                        : "neutral"
   611→                      : topic.sentiment.negative > topic.sentiment.neutral
   612→                        ? "negative"
   613→                        : "neutral";
   614→                  return (
   615→                    <tr
   616→                      key={topic.id}
   617→                      onClick={() => go("trends", { topicId: topic.id })}
   618→                      className="border-b border-border/50 last:border-0 hover:bg-accent cursor-pointer transition-colors"
   619→                    >
   620→                      <td className="px-4 py-2.5">
   621→                        <div className="flex items-center gap-2 min-w-0">
   622→                          {topic.velocity === "surging" && <Flame className="size-3 text-signal-orange shrink-0" />}
   623→                          <span className="text-xs font-medium text-foreground truncate max-w-40">{topic.label}</span>
   624→                          {topic.emerging && <Badge tone="orange">emerging</Badge>}
   625→                        </div>
   626→                        <div className="text-[10px] text-muted-foreground truncate max-w-44 mt-0.5">{topic.gloss}</div>
   627→                      </td>
   628→                      <td className="px-4 py-2.5 font-mono text-xs tnum text-foreground whitespace-nowrap">
   629→                        {fmtCompact(topic.baseVolume * 1.75)}
   630→                      </td>
   631→                      <td className="px-4 py-2.5 whitespace-nowrap">
   632→                        <Delta value={topic.change24h} />
   633→                      </td>
   634→                      <td className="px-4 py-2.5">
   635→                        <Badge tone={dom === "positive" ? "green" : dom === "negative" ? "red" : "slate"}>
   636→                          {dom}
   637→                        </Badge>
   638→                      </td>
   639→                      <td className="px-4 py-2.5 w-28">
   640→                        <ScoreBar value={topic.risk} showValue />
   641→                      </td>
   642→                      <td className="px-4 py-2.5">
   643→                        <Sparkline data={spark} color={CHART.orange} width={72} height={22} />
   644→                      </td>
   645→                    </tr>
   646→                  );
   647→                })}
   648→              </tbody>
   649→            </table>
   650→          </div>
   651→        </Panel>
   652→
   653→        <div className="xl:col-span-5 min-w-0">
   654→          <NetworkPreview />
   655→        </div>
   656→      </div>
   657→
   658→      {/* Module pipeline */}
   659→      <Panel title="Capability pipeline" icon={Cpu} sub="all modules nominal">
   660→        <ModuleStrip />
   661→      </Panel>
   662→
   663→      <div className="flex items-center gap-2 text-[10px] font-mono text-muted-foreground/60 px-1">
   664→        <Users className="size-3" />
   665→        Source: aggregated public posts (X, Telegram) · anonymised cohorts · window {windowLabel} ·
   666→        generated {relTime(NOW)} ago · prototype data
   667→      </div>
   668→    </div>
   669→  );
   670→}

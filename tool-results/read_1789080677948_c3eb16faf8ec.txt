   135→function NetworkPreview() {
   136→  const { go } = useApp();
   137→  const { filters } = useApp();
   138→  const { nodes, edges, communities } = useMemo(() => getNetwork(filters), [filters]);
   139→  const influencers = useMemo(() => getInfluencers(filters, 3), [filters]);
   140→
   141→  const W = 360;
   142→  const H = 210;
   143→  const layout = useMemo(() => {
   144→    const top = [...nodes].sort((a, b) => b.pageRank - a.pageRank).slice(0, 16);
   145→    const ids = new Set(top.map((n) => n.id));
   146→    const subEdges = edges.filter((e) => ids.has(e.source) && ids.has(e.target)).slice(0, 26);
   147→    // deterministic radial-ish layout
   148→    const placed = top.map((n, i) => {
   149→      const angle = (i / top.length) * Math.PI * 2 + (n.id.charCodeAt(4) % 10) * 0.06;
   150→      const radius = n.pageRank > 0.06 ? 0.28 : n.pageRank > 0.04 ? 0.55 : 0.82;
   151→      return {
   152→        ...n,
   153→        x: W / 2 + Math.cos(angle) * radius * (W / 2 - 18),
   154→        y: H / 2 + Math.sin(angle) * radius * (H / 2 - 14),
   155→      };
   156→    });
   157→    return { placed, subEdges };
   158→  }, [nodes, edges]);
   159→
   160→  const byId = new Map(layout.placed.map((n) => [n.id, n]));
   161→  const communityColor = (id: string) => communities.find((c) => c.id === id)?.color ?? "#5FA1C4";
   162→
   163→  return (
   164→    <Panel
   165→      title="Influence network"
   166→      sub="preview"
   167→      icon={Share2}
   168→      right={
   169→        <Button
   170→          variant="outline"
   171→          size="sm"
   172→          className="h-6 px-2 text-[10px] gap-1"
   173→          onClick={() => go("network")}
   174→        >
   175→          Open graph <ArrowUpRight className="size-3" />
   176→        </Button>
   177→      }
   178→    >
   179→      <button
   180→        type="button"
   181→        onClick={() => go("network")}
   182→        className="w-full block cursor-pointer group"
   183→        aria-label="Open full network analysis"
   184→      >
   185→        <svg viewBox={`0 0 ${W} ${H}`} className="w-full group-hover:opacity-90 transition-opacity">
   186→          {layout.subEdges.map((e, i) => {
   187→            const a = byId.get(e.source);
   188→            const b = byId.get(e.target);
   189→            if (!a || !b) return null;
   190→            return (
   191→              <line
   192→                key={i}
   193→                x1={a.x}
   194→                y1={a.y}
   195→                x2={b.x}
   196→                y2={b.y}
   197→                stroke="#232838"
   198→                strokeWidth={Math.max(0.6, e.weight / 6)}
   199→              />
   200→            );
   201→          })}
   202→          {layout.placed.map((n) => {
   203→            const r = 3 + Math.sqrt(n.pageRank) * 34;
   204→            const bot = n.botProb > 0.6;
   205→            return (
   206→              <g key={n.id}>
   207→                <circle
   208→                  cx={n.x}
   209→                  cy={n.y}
   210→                  r={r}
   211→                  fill={communityColor(n.communityId)}
   212→                  fillOpacity={0.85}
   213→                  stroke={bot ? CHART.red : "none"}
   214→                  strokeWidth={bot ? 1.4 : 0}
   215→                />
   216→                {n.botProb > 0.6 && (
   217→                  <circle cx={n.x} cy={n.y} r={r + 2.5} fill="none" stroke={CHART.red} strokeOpacity={0.35} />
   218→                )}
   219→              </g>
   220→            );
   221→          })}
   222→        </svg>
   223→      </button>
   224→      <div className="mt-3 space-y-2">
   225→        <Taxonomy>Top influencers</Taxonomy>
   226→        {influencers.map((inf) => (
   227→          <div key={inf.nodeId} className="flex items-center gap-2 text-xs">
   228→            <span className="size-2 rounded-[2px] shrink-0" style={{ background: inf.communityColor }} />
   229→            <span className="font-mono text-[11px] text-foreground truncate">{inf.handle}</span>
   230→            <span className="ml-auto font-mono text-[10px] tnum text-muted-foreground shrink-0">
   231→              PR {inf.pageRank.toFixed(3)}
   232→            </span>
   233→            <span className="font-mono text-[10px] tnum text-muted-foreground shrink-0 w-12 text-right">
   234→              {fmtCompact(inf.reach)}
   235→            </span>
   236→          </div>
   237→        ))}
   238→      </div>
   239→    </Panel>
   240→  );
   241→}

   286→export function OverviewScreen() {
   287→  const { filters, go, setRange, alertsFeed } = useApp();
   288→  const ready = useRefresh("overview");
   289→
   290→  const kpis = useMemo(() => getKpis(filters), [filters]);
   291→  const series = useMemo(() => getVolumeSeries(filters), [filters]);
   292→  /* 30-day corpus heat strip — always 30d context regardless of active range */
   293→  const heatDays = useMemo<HeatDay[]>(
   294→    () => getVolumeSeries({ ...filters, range: "30d" }).map((p) => ({ t: p.t, total: p.total, spike: p.spike, label: p.label })),
   295→    [filters]
   296→  );
   297→  const onHeatDay = (day: HeatDay) => {
   298→    setRange("30d");
   299→    go("trends");
   300→    toast(`Corpus view: ${day.label}`, {
   301→      description: `${fmtFull(day.total)} posts on that day — 30-day window loaded in Trend Explorer.`,
   302→    });
   303→  };
   304→  const topics = useMemo(
   305→    () =>
   306→      effectiveTopics(filters)
   307→        .map((t) => ({ ...t, spark: getTopicSeries(t.topic, filters).map((p) => p.total) }))
   308→        .sort((a, b) => b.topic.change24h - a.topic.change24h)
   309→        .slice(0, 7),
   310→    [filters]
   311→  );
   312→  const chartTheme = useChartTheme();
   313→
   314→  /* ---- temporal replay (v0.13 time machine) ----
   315→     cursorIdx = null → live full window; otherwise the playhead index.
   316→     Playback = a self-rescheduling timeout (recreated per tick because
   317→     cursorIdx is a dependency, so the closure always sees fresh state);
   318→     ALL setState happens inside the async tick callback — lint-clean. */
   319→  const [cursorIdx, setCursorIdx] = useState<number | null>(null);
   320→  const [playing, setPlaying] = useState(false);
   321→  const [speed, setSpeed] = useState(1);
   322→
   323→  const stepMs = Math.max(120, Math.round(14_000 / Math.max(2, series.length) / speed));
   324→  useEffect(() => {
   325→    if (!playing || series.length < 2) return;
   326→    const cur = cursorIdx ?? series.length - 1;
   327→    const id = setTimeout(() => {
   328→      if (cur >= series.length - 1) {
   329→        /* swept to the live edge — hand back to the live window */
   330→        setPlaying(false);
   331→        setCursorIdx(null);
   332→        return;
   333→      }
   334→      setCursorIdx(cur + 1);
   335→    }, stepMs);
   336→    return () => clearTimeout(id);
   337→  }, [playing, cursorIdx, series.length, stepMs]);
   338→
   339→  /* keyboard seek keeps playing: TimeMachine calls onCursor directly
   340→     (no onPlaying(false)) — the timer picks up from the new position */
   341→  const onSeekKeepPlay = (i: number | null) => setCursorIdx(i);
   342→
   343→  const isLive = cursorIdx == null;
   344→  const effCursor = Math.min(cursorIdx ?? series.length - 1, series.length - 1);
   345→  /* every artefact below renders from `view` — the as-of slice */
   346→  const view = useMemo(
   347→    () => (isLive ? series : series.slice(0, effCursor + 1)),
   348→    [series, isLive, effCursor]
   349→  );
   350→  const viewEnd = view[view.length - 1];
   351→
   352→  const totalVol = useMemo(() => series.reduce((a, p) => a + p.total, 0) || 1, [series]);
   353→  const viewVol = useMemo(() => view.reduce((a, p) => a + p.total, 0), [view]);
   354→  /* as-of KPI derivations — real arithmetic over the sliced corpus */
   355→  const postsAsOf = Math.round(kpis.postsTracked * (viewVol / totalVol));
   356→  const sentimentAsOf = useMemo(() => {
   357→    const pos = view.reduce((a, p) => a + p.positive, 0);
   358→    const neg = view.reduce((a, p) => a + p.negative, 0);
   359→    const tot = view.reduce((a, p) => a + p.total, 0) || 1;
   360→    return (pos - neg) / tot;
   361→  }, [view]);
   362→  const narrativesAsOf = useMemo(
   363→    () =>
   364→      topics.filter((t) => {
   365→        const prefix = t.spark.slice(0, effCursor + 1);
   366→        const full = t.spark.reduce((a, b) => a + b, 0) || 1;
   367→        return prefix.reduce((a, b) => a + b, 0) / full >= 0.12;
   368→      }).length,
   369→    [topics, effCursor]
   370→  );
   371→  const highRiskAsOf = useMemo(() => {
   372→    if (isLive) return kpis.highRiskAlerts;
   373→    const edge = viewEnd?.t ?? 0;
   374→    return alertsFeed.filter(
   375→      (a) => a.t <= edge && a.status === "New" && (a.severity === "critical" || a.severity === "high")
   376→    ).length;
   377→  }, [alertsFeed, isLive, viewEnd, kpis.highRiskAlerts]);
   378→
   379→  /* chart payload: stacked sentiment up to the playhead, ghost of the
   380→     still-upcoming window after it (null-padded so recharts breaks both) */
   381→  const chartData = useMemo(
   382→    () =>
   383→      series.map((p, i) =>
   384→        i <= effCursor
   385→          ? { ...p, ghost: null as number | null }
   386→          : { ...p, positive: null, neutral: null, negative: null, baseline: null, ghost: p.total }
   387→      ),
   388→    [series, effCursor]
   389→  );
   390→
   391→  /* trending rows during replay: only narratives that have actually
   392→     emerged by the playhead, ranked by their as-of growth rate, with
   393→     volume and spark truncated to the arrived prefix */
   394→  const trendingAsOf = useMemo(() => {
   395→    if (isLive) return topics.map((t) => ({ ...t, share: 1, asOfChange: t.topic.change24h, spark: t.spark }));
   396→    const rows = topics
   397→      .map((t) => {
   398→        const prefix = t.spark.slice(0, effCursor + 1);
   399→        const full = t.spark.reduce((a, b) => a + b, 0) || 1;
   400→        const share = prefix.reduce((a, b) => a + b, 0) / full;
   401→        const n = prefix.length;
   402→        const last3 = prefix.slice(-3).reduce((a, b) => a + b, 0) / Math.max(1, Math.min(3, n));
   403→        const prev3 = prefix.slice(Math.max(0, n - 6), Math.max(0, n - 3));
   404→        const prevAvg = prev3.length ? prev3.reduce((a, b) => a + b, 0) / prev3.length : last3;
   405→        const asOfChange = prevAvg > 0 ? ((last3 - prevAvg) / prevAvg) * 100 : t.topic.change24h;
   406→        return { ...t, spark: prefix, share, asOfChange };
   407→      })
   408→      .filter((t) => t.share >= 0.12)
   409→      .sort((a, b) => b.asOfChange - a.asOfChange);
   410→    return rows.slice(0, 7);
   411→  }, [topics, isLive, effCursor]);
   412→
   413→  if (!ready) {
   414→    return (
   415→      <div className="space-y-4 animate-in fade-in duration-200">
   416→        <KpiRowSkeleton />
   417→        <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
   418→          <PanelSkeleton className="xl:col-span-8" height="h-72" />
   419→          <PanelSkeleton className="xl:col-span-4" height="h-72" />
   420→        </div>
   421→        <PanelSkeleton height="h-48" />
   422→      </div>
   423→    );
   424→  }
   425→
   426→  const windowLabel = WINDOW_LABEL[filters.range] ?? "window";
   427→  const sentimentToneOfKpi = sentimentTone(kpis.avgSentiment);
   428→  const sentimentToneAsOf = sentimentTone(sentimentAsOf);
   429→  const replayLabel = viewEnd?.label ?? "—";
   430→
   431→  return (
   432→    <div className="space-y-4 animate-in fade-in slide-in-from-bottom-2 duration-300">
   433→      <ScreenHeader
   434→        kicker="MODULE 01 // INTELLIGENCE FUSION"
   435→        title="Mission Control"
   436→        description={`Unified signal picture across X and Telegram for the ${windowLabel} — volume, narratives, influence, bot activity and misinformation risk in a single operating view.`}
   437→        right={
   438→          <div className="flex items-center gap-2 flex-wrap">
   439→            <Badge tone="cyan" dot>
   440→              {filters.platform === "all" ? "X + Telegram" : filters.platform === "x" ? "X only" : "Telegram only"}
   441→            </Badge>
   442→            <Badge tone={sentimentToneOfKpi} dot>
   443→              net {fmtNet(kpis.avgSentiment)}
   444→            </Badge>
   445→            {!isLive && (
   446→              <Badge tone="orange" dot>
   447→                <History className="size-2.5" /> replay · as of {replayLabel}
   448→              </Badge>
   449→            )}
   450→            <Button
   451→              variant="outline"
   452→              size="sm"
   453→              className="h-7 text-[11px] gap-1.5"
   454→              onClick={() => {
   455→                toast("Fusion snapshot queued", {
   456→                  description: "Intelligence summary attached to the daily briefing (mock).",
   457→                });
   458→              }}
   459→            >
   460→              <Layers className="size-3.5" /> Snapshot
   461→            </Button>
   462→          </div>
   463→        }
   464→      />
   465→
   466→      {/* KPI row */}
   467→      <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4" data-tour="kpi-row">
   468→        <KpiCard
   469→          label="Posts tracked"
   470→          value={fmtCompact(postsAsOf)}
   471→          icon={MessageSquare}
   472→          tone="cyan"
   473→          delta={12.4}
   474→          spark={view.map((p) => p.total)}
   475→          sparkColor={CHART.cyan}
   476→          footnote={
   477→            isLive ? (
   478→              <>
   479→                <span className="text-signal-cyan">X {Math.round(kpis.xShare * 100)}%</span>
   480→                <span className="text-border">/</span>
   481→                <span className="text-signal-green">TG {Math.round(kpis.telegramShare * 100)}%</span>
   482→              </>
   483→            ) : (
   484→              <span className="text-signal-orange">as of {replayLabel}</span>
   485→            )
   486→          }
   487→        />
   488→        <KpiCard
   489→          label="Active narratives"
   490→          value={String(narrativesAsOf)}
   491→          icon={GitBranch}
   492→          tone="orange"
   493→          delta={8.1}
   494→          spark={view.map((p) => Math.max(1, Math.round(p.total / 900)))}
   495→          sparkColor={CHART.orange}
   496→          footnote={<span>{isLive ? "3 emerging · 2 coordinated" : `${Math.round((viewVol / totalVol) * 100)}% of window volume`}</span>}
   497→        />

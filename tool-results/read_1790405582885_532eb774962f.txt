    92→/** Analyst notebook — free-form annotation attached to a claim, persisted
    93→ *  in the session store (survives reload; shared across dossiers).
    94→ *  Remounts per claim (keyed by claim id) so the draft initialises cleanly
    95→ *  without setState-in-effect. Saves are debounced 500ms + on blur. */
    96→function ClaimNotebook({ claimId }: { claimId: string }) {
    97→  const { claimNotes, setClaimNote } = useApp();
    98→  const saved = claimNotes[claimId]?.text ?? "";
    99→  const [draft, setDraft] = useState(saved);
   100→  const [dirty, setDirty] = useState(false);
   101→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   102→
   103→  const commit = (text: string) => {
   104→    if (debounceRef.current) clearTimeout(debounceRef.current);
   105→    setClaimNote(claimId, text);
   106→    setDirty(false);
   107→  };
   108→
   109→  const onChange = (v: string) => {
   110→    setDraft(v);
   111→    setDirty(true);
   112→    if (debounceRef.current) clearTimeout(debounceRef.current);
   113→    debounceRef.current = setTimeout(() => commit(v), 500);
   114→  };
   115→
   116→  /* flush pending edits if the component unmounts mid-debounce */
   117→  useEffect(
   118→    () => () => {
   119→      if (debounceRef.current) clearTimeout(debounceRef.current);
   120→    },
   121→    []
   122→  );
   123→
   124→  const updatedAt = claimNotes[claimId]?.updatedAt;
   125→
   126→  return (
   127→    <div className="mt-4 pt-3 border-t border-border/60">
   128→      <div className="flex items-center gap-2 mb-2">
   129→        <NotebookPen className="size-3.5 text-primary" />
   130→        <Taxonomy>Analyst notebook</Taxonomy>
   131→        <span className="ml-auto font-mono text-[9px] text-muted-foreground/70">
   132→          {dirty ? "saving…" : updatedAt ? `saved ${relTime(updatedAt, NOW)} ago` : "empty"}
   133→        </span>
   134→        {draft.trim() !== "" && (
   135→          <span
   136→            className={cn(
   137→              "inline-flex items-center gap-1 font-mono text-[9px]",
   138→              dirty ? "text-muted-foreground" : "text-signal-green"
   139→            )}
   140→          >
   141→            <Check className="size-2.5" />
   142→            {dirty ? "draft" : "synced"}
   143→          </span>
   144→        )}
   145→      </div>
   146→      <Textarea
   147→        value={draft}
   148→        onChange={(e) => onChange(e.target.value)}
   149→        onBlur={() => dirty && commit(draft)}
   150→        placeholder="Triage notes, escalation decisions, cross-references… (auto-saves, persists across sessions)"
   151→        className="min-h-20 text-xs bg-background font-mono leading-relaxed resize-y"
   152→        maxLength={2000}
   153→        aria-label={`Analyst notes for claim ${claimId}`}
   154→      />
   155→      <div className="flex items-center justify-between mt-1.5">
   156→        <span className="font-mono text-[9px] tnum text-muted-foreground/60">{draft.length}/2000</span>
   157→        {draft !== saved && (
   158→          <button
   159→            type="button"
   160→            onClick={() => commit(draft)}
   161→            className="text-[10px] font-mono uppercase tracking-wider text-primary hover:text-primary/80 transition-colors cursor-pointer"
   162→          >
   163→            Save now
   164→          </button>
   165→        )}
   166→      </div>
   167→    </div>
   168→  );
   169→}

   401→export function MisinfoScreen() {
   402→  const { filters, selectedClaimId, setSelectedClaimId, claimNotes } = useApp();
   403→  const ready = useRefresh("misinfo");
   404→
   405→  const claims = useMemo(() => getClaims(filters), [filters]);
   406→  const [riskFilter, setRiskFilter] = useState<RiskFilter>("all");
   407→  const [statusFilter, setStatusFilter] = useState<StatusFilter>("all");
   408→  /* dossier selection lives in the global bus so the ⌘K palette and
   409→     deep-links (#/misinfo/claim:CLM-004) can open a dossier directly. */
   410→  const selectedId = selectedClaimId;
   411→  const setSelectedId = setSelectedClaimId;
   412→
   413→  const sorted = useMemo(() => [...claims].sort((a, b) => b.risk - a.risk), [claims]);
   414→  const riskTest = RISK_FILTERS.find((f) => f.id === riskFilter)?.test ?? (() => true);
   415→  const filtered = useMemo(
   416→    () => sorted.filter((c) => riskTest(c.risk) && (statusFilter === "all" || c.status === statusFilter)),
   417→    [sorted, riskTest, statusFilter]
   418→  );
   419→
   420→  /* selection: explicit pick, else highest-risk claim of the current bank */
   421→  const selected = useMemo(
   422→    () => claims.find((c) => c.id === selectedId) ?? filtered[0] ?? null,
   423→    [claims, selectedId, filtered]
   424→  );
   425→
   426→  /* Deep-link / palette focus: when a claim becomes selected, scroll it into
   427→   * view inside the extraction list and flash an outline pulse. DOM-only side
   428→   * effect (no setState) — fires for user clicks and programmatic jumps alike. */
   429→  useEffect(() => {
   430→    if (!selectedId) return;
   431→    const el = document.querySelector<HTMLElement>(`[data-claim-id="${CSS.escape(selectedId)}"]`);
   432→    if (!el) return;
   433→    el.scrollIntoView({ behavior: "smooth", block: "nearest" });
   434→    el.classList.remove("claim-flash");
   435→    /* restart the animation even when re-selecting the same claim */
   436→    void el.offsetWidth;
   437→    el.classList.add("claim-flash");
   438→    const t = setTimeout(() => el.classList.remove("claim-flash"), 1_800);
   439→    return () => clearTimeout(t);
   440→  }, [selectedId]);
   441→
   442→  /* KPIs */
   443→  const total = claims.length;
   444→  const falseDisputed = claims.filter((c) => c.status === "False" || c.status === "Disputed");
   445→  const unverified = claims.filter((c) => c.status === "Unverified");
   446→  const avgRisk = total ? claims.reduce((a, c) => a + c.risk, 0) / total : 0;
   447→  const botCorrelated = claims.filter((c) => c.botCorrelation >= 0.5);
   448→
   449→  const windowLabel = WINDOW_LABEL[filters.range] ?? "window";
   450→  const platformLabel =
   451→    filters.platform === "all" ? "X + Telegram" : filters.platform === "x" ? "X only" : "Telegram only";

   453→  const botSpark = useMemo(() => claims.map((c) => c.botCorrelation).sort((a, b) => a - b), [claims]);
   454→
   455→  if (!ready) {
   456→    return (
   457→      <div className="space-y-4 animate-in fade-in duration-200">
   458→        <KpiRowSkeleton />
   459→        <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
   460→          <PanelSkeleton className="xl:col-span-7" height="h-[520px]" />
   461→          <PanelSkeleton className="xl:col-span-5" height="h-[520px]" />
   462→        </div>
   463→      </div>
   464→    );
   465→  }
   466→
   467→  return (
   468→    <div className="space-y-4 animate-in fade-in duration-300">
   469→      <ScreenHeader
   470→        kicker="MODULE 07 // MISINFORMATION RADAR"
   471→        title="Misinformation Radar"
   472→        description={`Claims extracted from the monitored stream, cross-referenced against fact-check corpora and tracked as they propagate across X and Telegram — ${windowLabel}. Select a claim to open its dossier.`}
   473→        right={
   474→          <div className="flex items-center gap-2">
   475→            <Badge tone="cyan" dot>
   476→              {platformLabel}
   477→            </Badge>
   478→            <Badge tone={riskTone(avgRisk)} dot>
   479→              avg risk {total ? avgRisk.toFixed(2) : "—"}
   480→            </Badge>
   481→            <Button
   482→              variant="outline"
   483→              size="sm"
   484→              className="h-7 text-[11px] gap-1.5"
   485→              onClick={() => {
   486→                const top = sorted[0];
   487→                if (!top) return;
   488→                toast("Claim escalated", {
   489→                  description: `${top.id} routed to the fact-check escalation queue.`,
   490→                });
   491→              }}
   492→            >
   493→              <ShieldAlert className="size-3.5" /> Escalate top claim
   494→            </Button>
   495→          </div>
   496→        }
   497→      />
   498→
   499→      {/* KPI row */}
   500→      <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
   501→        <KpiCard
   502→          label="Claims tracked"
   503→          value={String(total)}
   504→          icon={FileSearch}
   505→          tone="cyan"
   506→          delta={11.1}
   507→          spark={riskSpark}
   508→          sparkColor={CHART.cyan}
   509→          footnote={
   510→            <span>
   511→              {falseDisputed.length} false/disputed · {unverified.length} unverified
   512→            </span>
   513→          }
   514→        />
   515→        <KpiCard
   516→          label="False + disputed rate"
